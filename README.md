/*
 * Error ≠ failure; it's debugging data.
 * finally: don't power off.
 */
public class Philosophy {
    public static void main(String[] args) {
        try {
            live();
        } catch (Exception e) {
            debug(e); // it's debugging data
        } finally {
            keepAlive(); // don't power off
        }
    }
}
